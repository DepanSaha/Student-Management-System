----USER TABLE----
CREATE TABLE USER(
USERID VARCHAR(20) PRIMARY KEY, 
PASSWORD VARCHAR(20), 
ROLE VARCHAR(9));

----STUDENT_MASTER TABLE----
CREATE TABLE STUDENT_MASTER(
STUDENT_ID VARCHAR(15) PRIMARY KEY, 
NAME VARCHAR(20), 
FATHER_NAME VARCHAR(20), 
GENDER VARCHAR(6), 
ADDRESS VARCHAR(50), 
DOB DATE, 
PHONE VARCHAR(12), 
EMAIL VARCHAR(30), 
COURSE VARCHAR(5), 
SEMESTERS CHAR(1));

----PROFESSOR_MASTER TABLE----
CREATE TABLE PROFESSOR_MASTER(
PROFESSOR_ID VARCHAR(13) PRIMARY KEY, 
NAME VARCHAR(10), 
ADDRESS VARCHAR(50), 
GENDER VARCHAR(6), 
PHONE VARCHAR(12), 
EMAIL VARCHAR(30), 
DOB DATE, 
DOJ DATE);

----PROFESSOR_DEGREE----
CREATE TABLE PROFESSOR_DEGREE(
PROFESSOR_ID VARCHAR(13), 
DEGREE VARCHAR(10), 
PRIMARY KEY (PROFESSOR_ID, DEGREE), 
FOREIGN KEY(PROFESSOR_ID) REFERENCES PROFESSOR_MASTER(PROFESSOR_ID) ON DELETE CASCADE);

----STUDENT_GRADE TABLE-----
CREATE TABLE STUDENT_GRADE(
STUDENT_ID VARCHAR(20), 
SEMESTER CHAR(1), 
GRADE CHAR(1), 
PRIMARY KEY(STUDENT_ID, SEMESTER), 
FOREIGN KEY(STUDENT_ID) REFERENCES STUDENT_MASTER(STUDENT_ID) ON DELETE CASCADE);
